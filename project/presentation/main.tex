\begin{frame}
\frametitle{Data Profiling — Dependency Discovery}
  \begin{items}
    \begin{oitem}{\litem}
      Data Profiling Stack: \\
      \sitem {\footnotesize Metanome Platform used for all dependency discover algorithms.}
    \end{oitem}
    \begin{oitem}{\litem}
      Unique Column Combinations: Using HyUCC algorithm. \\
      \sitem {\footnotesize Discovered only single-column candidate keys across tables: \inlinecode{ogc_fid}, \inlinecode{gml_id}, \inlinecode{objectid}.} \\
      \sitem {\footnotesize Selected \inlinecode{objectid} as primary key due to global uniqueness across tables.}
    \end{oitem}
    \begin{oitem}{\litem}
      Functional Dependencies: Using HyFD algorithm. \\
      \sitem {\footnotesize \inlinecode{code} $\rightarrow$ \{\inlinecode{bez_d}, \inlinecode{bez_i}\} Infrastructure type names redundant in asset tables.} \\
      \sitem {\footnotesize \inlinecode{code} $\rightarrow$ \{\inlinecode{gefahr}, \inlinecode{pericolo}\} Hazard classifications redundant.} \\
      \sitem {\footnotesize \inlinecode{distr_code} $\rightarrow$ \{\inlinecode{distr_de}, \inlinecode{distr_it}\} District names redundant in municipal table.} \\
    \end{oitem}
    \begin{oitem}{\litem}
      Inclusion Dependencies: Using BINDER algorithm: \\
      \sitem {\footnotesize Confirmed foreign keys \inlinecode{istat_code} are strictly included in \inlinecode{Municipality.istat_code}.}
    \end{oitem}
  \end{items}
\end{frame}

\begin{frame}
\frametitle{Relational Database Schema Design}
  \begin{items}
    \begin{oitem}{\litem}
      Normalization and Entity Extraction: \\
      \sitem {\footnotesize Extracted 4 relations to avoid duplication: \\InfrastructureType, DangerLevel, HazardProcess, District} \\
      \sitem {\footnotesize Maintained 5 core spatial relations: \\InfrastructureLine, InfrastructureNode, LandslideHazard, AvalancheHazard, Municipality}
    \end{oitem}
    \begin{oitem}{\litem}
      Schema Constraints and Data Types: \\
      \sitem {\footnotesize Primary keys enforced on \inlinecode{code} and \inlinecode{istat_code}.} \\
      \sitem {\footnotesize Foreign keys enforced for municipality links and normalized tables.} \\
      \sitem {\footnotesize Transformed trailing float identifiers e.g., \inlinecode{21001.0}, to standard \inlinecode{INTEGER}.} \\
      \sitem {\footnotesize Geometries mapped to PostGIS \inlinecode{GEOMETRY} type.}
    \end{oitem}
    \begin{oitem}{\litem}
      Attribute Cleanup: \\
      \sitem {\footnotesize Dropped empty and constant fields: \inlinecode{blp}, \inlinecode{codealt}, \inlinecode{status_gzp}, \inlinecode{id_phen}.} \\
      \sitem {\footnotesize Dropped UI-only and survey fields: \inlinecode{nda_text_*}, \inlinecode{nda_link_*}, \inlinecode{x_label}, \inlinecode{y_label}, \inlinecode{bearbeitungstiefe}.} \\
      \sitem {\footnotesize Omitted derived metrics, \inlinecode{shape_len} and \inlinecode{area}, to calculate them dynamically.}
    \end{oitem}
  \end{items}
\end{frame}

\begin{frame}
\frametitle{Domain Ontology Modeling}
  \begin{items}
    \begin{oitem}{\litem}
      Ontology Architecture: \\
      \sitem {\footnotesize 18 classes, 5 object properties, and 20 data properties.} \\
      \sitem {\footnotesize \inlinecode{GeometricObject} $\rightarrow$ (disjoint) \inlinecode{Infrastructure}, \inlinecode{HazardZone}, \inlinecode{Municipality}.}
    \end{oitem}
    \begin{oitem}{\litem}
      Specialized Subclasses: \\
      \sitem {\footnotesize Structural (\inlinecode{Line}, \inlinecode{Node}) and Thematic (\inlinecode{Energy}, \inlinecode{Water}, \inlinecode{Waste}, \inlinecode{Communication}, \inlinecode{Electricity}).} \\
      \sitem {\footnotesize Type (\inlinecode{LandslideZone}, \inlinecode{AvalancheZone}) and Severity (\inlinecode{VeryHigh}, \inlinecode{High}, \inlinecode{Medium}, \inlinecode{Low})}
    \end{oitem}
    \begin{oitem}{\litem}
      Key Object Properties: \\
      \sitem {\footnotesize \inlinecode{intersectsWith}: Symmetric topological overlap.} \\
      \sitem {\footnotesize \inlinecode{isExposedTo} $\sqsubseteq$ \inlinecode{intersectsWith}: Links \inlinecode{Infrastructure} directly to \inlinecode{HazardZone}.} \\
      \sitem {\footnotesize \inlinecode{infrastructureIn}, \inlinecode{hazardZoneIn}, \inlinecode{belongsTo}: Administrative links.}
    \end{oitem}
    \begin{oitem}{\litem}
      Data Properties: \\
      \sitem {\footnotesize Encapsulate WKT geometry strings, area and length attributes, localized names, and codes.}
    \end{oitem}
  \end{items}
\end{frame}
